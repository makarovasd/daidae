<script src="https://cdn.goeasy.io/goeasy-2.5.0.min.js"></script>
<script>
const GOEASY_APPKEY = "BC-c21dce55a19c4231b5dd036ee81b3211";
const goeasy = new GoEasy({
  appkey: GOEASY_APPKEY,
  onConnected: ()=>{
    document.getElementById("tip").textContent="已连接，可以同步";
  },
  onDisconnected: ()=>{
    document.getElementById("tip").textContent="断开，正在重连";
  }
});
const channel = "room001";

const gridSize = 8;
let img = new Image();
img.src = "source.png"; // 这里改成你仓库里的图片名字
let cellW, cellH;
let state = {
  order: [],
  active: []
};

function render(){
  const gridEl = document.getElementById("grid");
  gridEl.innerHTML = "";
  gridEl.style.gridTemplateColumns = `repeat(${gridSize}, ${cellW}px)`;
  state.order.forEach((idx, i)=>{
    const div = document.createElement("div");
    div.className = "cell" + (state.active.includes(i)?" active":"");
    div.style.width = cellW+"px";
    div.style.height = cellH+"px";
    div.style.backgroundImage = `url(source.png)`;
    const rx = (idx % gridSize) * cellW;
    const ry = Math.floor(idx / gridSize) * cellH;
    div.style.backgroundPosition = `-${rx}px -${ry}px`;
    div.onclick = ()=>{
      const newActive = [...state.active];
      const p = newActive.indexOf(i);
      if(p>=0) newActive.splice(p,1);
      else newActive.push(i);
      const newData = {order:state.order, active:newActive};
      goeasy.publish({
        channel: channel,
        message: JSON.stringify(newData)
      })
    }
    gridEl.appendChild(div);
  })
}

goeasy.subscribe({
  channel: channel,
  onMessage: function(message){
    state = JSON.parse(message.content);
    render();
  }
});

img.onload = ()=>{
  cellW = img.width / gridSize;
  cellH = img.height / gridSize;
  state.order = Array.from({length:gridSize*gridSize},(_,i)=>i);
  state.active = [];
  render();
}

document.getElementById("shuffleBtn").onclick = ()=>{
  const order = [...state.order];
  for(let i=order.length-1;i>0;i--){
    const j = Math.floor(Math.random()*(i+1));
    [order[i],order[j]] = [order[j],order[i]];
  }
  const newData = {order, active:state.active};
  goeasy.publish({
    channel: channel,
    message: JSON.stringify(newData)
  })
}
</script>
